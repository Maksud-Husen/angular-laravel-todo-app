  location / {
              try_files $uri $uri/ /index.php?$query_string;          - "this is for so that when we try to open and if file not found it be passed to index.php"
          }
  ------------------------------------------------------------------------------------------------------------------------------------------------------------------
  location ~ \.php$ {
              fastcgi_pass php:9000;
              fastcgi_index index.php;
              fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
              include fastcgi_params;
          }

  fastcgi_pass - it forward the request commping from the user to the php running in the port 9000
  fastcgi_param - this line script sets the path for the PHP-FPM to execute.
  include fastcgi_params - it inclues  standard FastCGI configuration variables.
  ---------------------------------------------------------------------------------------------------------------------------------------------------------------
  location ~ /\.ht {
              deny all;                                               - " this whole commad denys any accesss to .ht files which are sensative for the program"
          }

  ---

  - name: Install MySQL service
    apt:
      name: mysql-server
      state: present
    when: "'mysql-server' not in ansible_facts.packages"               - "this part is specifically used to install the mysql in and when command is for it the project is not in the device then only run the program" 

  - name: Install Python MySQL client
    apt:
      name: python3-mysqldb
      state: present
    when: "'python3-mysqldb' not in ansible_facts.packages"            - "Similarly we will be insatalling the python3-mysqldb . So, that we can Create user later on in ther server "

  - name: Start and enable Mysql

      name: mysql
      state: started 
      enabled: true                                                    - "This command is for starting and enabling the mysql. So, that we can make and run the database"
      -----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
  then in roles/db/tasks/service.yml
  ---
  - name: Create database 
    mysql_db:
      name: todoapp                                                    - "In this we are creating an Database named as todoapp"
      state: present
      
  - name: create user
    mysql_user:
      name: homestead
      password: todoapp786@
      host: <write your app server IP here>                            - "This command is specifically for creating user for which we need the listed item like name,password,host_ip,
      priv: 'todoapp.*:ALL'                                              priv(for what to give the privalage)"
      state: present

  - name: firewall rule to allow app server to connect to db server
    community.general.ufw:
      rule: allow
      port: 3306                                                       - "This comand is used for the firewall to allow the server to give access for app server with its ip and for which port to allow to get access"
      proto: tcp                                                         as given in this command (it is giving access to app server to be able to get access to port 3306 for database)
      from_ip: <write your app server IP here>                        
                                                          
  - name: restart mysql service
    service:
      name: mysql                                                     - "After configurating every thing in the mysql we need to restart it as i have done here"
      state: restarted

  Step 2:- writting the yml file for web server
  roles/web/tasks/rm_container.yml

  ---
  - name: Stop all running containers
    shell: docker stop $(docker ps -aq)
    ignore_errors: yes


  - name: Removing old containers
    shell: docker rm -f $(docker ps -aq)                             - "This part is important to deploy the app multiple time in same server for testing 
    ignore_errors: yes                                               because it will remove every old container and images so that it don't create any issue from old container and images "

  - name: Removing old images
    shell: docker rmi -f $(docker images -aq) || true                - Note:- if there is non container and images it way provide error while running the program for thet we can ignore it 
    ignore_errors: yes                                                        for which i have given the ignore_errors: yes command
  ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------

  then roles/web/tasks/install.yml
  ---
  - name: Gather installed package facts
    package_facts:
      manager: auto

  - name: Install Nginx

      name: nginx
      state: present
    when: "'nginx' not in ansible_facts.packages"

  - name: Install Docker and Docker Compose

      name:
        - docker.io                                            - Notes:- if there are multiple package to install we can use this specific way to run the loop to install each package one by one as shown in left

      state: present
    when: "'docker.io' not in ansible_facts.packages and 'docker-compose' not in ansible_facts.packages"

  - name: Start and enable Nginx

      name: nginx

      enabled: true
